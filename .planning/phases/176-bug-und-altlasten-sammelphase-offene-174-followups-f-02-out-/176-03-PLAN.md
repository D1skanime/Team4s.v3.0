       string_agg(acr.role_code, ',') as roles

LEFT JOIN anime_contribution_roles acr ON acr.anime_contribution_id = ac.id
WHERE m.nickname='timer' AND ac.anime_id=1 AND ac.fansub_group_id=1
GROUP BY ac.id, ac.release_version_id, ac.created_at, ac.created_by, ac.updated_by ORDER BY ac.id;

-- systemic correlation check: how many anime-wide confirmed rows have a matching set of

SELECT count(*) FROM wide w JOIN scoped s USING (member_id, anime_id, fansub_group_id);

SELECT snapshot_mode, count(*) FROM release_crew_snapshots GROUP BY 1;
