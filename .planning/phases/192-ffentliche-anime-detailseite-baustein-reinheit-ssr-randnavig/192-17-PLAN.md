```ts
const [hoverDirection, setHoverDirection] = useState<Direction | null>(null)
function withPreviewHandlers(item, direction) {
  return { ...item, onMouseEnter: () => setHoverDirection(direction), onMouseLeave: () => setHoverDirection((c) => c === direction ? null : c), onFocus: () => setHoverDirection(direction), onBlur: () => setHoverDirection((c) => c === direction ? null : c), onTouchStart: () => setHoverDirection(direction) }
}
const previewAnime = hoverDirection === 'prev' ? previousAnime : hoverDirection === 'next' ? nextAnime : null
```

`useFansubGroupSwitch`/`useWatchlistAction` already established.
