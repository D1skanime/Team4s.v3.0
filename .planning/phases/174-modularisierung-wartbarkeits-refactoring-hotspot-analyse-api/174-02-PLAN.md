```typescript
export function resolveApiUrl(value?: string): string {
  const trimmed = (value || "").trim();
  if (!trimmed) { return ""; }
  if (trimmed.startsWith("http://") || trimmed.startsWith("https://")) {
    const loopbackMediaUrl = resolveLoopbackMediaUrl(trimmed);
    if (loopbackMediaUrl) { return loopbackMediaUrl; }
    return trimmed;
  }
  if (trimmed.startsWith("/api/")) { return resolvePublicApiUrl(trimmed); }
  return trimmed;
}
```
