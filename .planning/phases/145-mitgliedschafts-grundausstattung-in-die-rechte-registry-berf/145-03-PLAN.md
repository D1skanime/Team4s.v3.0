export function roleKindLabel(role: RoleEntry): string {
  const isEditable = role.capability_editable !== false
  const isAssignable = role.assignable === true
  if (role.role_kind === 'global_app_role') return 'Globale App-Rolle'

}

- `145-UI-SPEC.md`'s Interaction Contract item 3 claims "the existing fallback already renders –" for
  the pseudo-role's holder-count text — this is INACCURATE for `RoleDetailPanel.tsx`'s
  `holderCountText()` (checked directly against the real source): that function only special-cases
  `role.role_kind === 'global_app_role'`; for every other role it falls straight through to
  `` `${holders.length} Inhaber` `` with NO "–" fallback at all (that fallback only exists in
  `RoleRail.tsx`'s separate `rowCountText()`, used for a different piece of UI — the rail row, not the
  detail-panel header). Since `RolesClient.tsx`'s `loadHolders` is skipped for the pseudo-role (per
  Interaction Contract item 4) and `holders` state is never reset on role change, selecting the
  pseudo-role would otherwise display a STALE holder count left over from whichever role was
  previously selected, in the subject-header line next to the role name. This task adds a
  `role.role_kind === 'reserved_baseline'` branch to `holderCountText()` returning `'–'` — a genuine
  bug-fix found by reading the real code, not a request to reinterpret the UI-SPEC's design intent.

  `'@/components/ui/Switch'` (direct module paths, not the `'@/components/ui'` barrel) — add the new
  `Button` import the same way: `import { Button } from '@/components/ui/Button'`.
- `RoleCapabilityImpactPreviewModal.tsx`, `GuidedRevokeFlow.tsx`, and `userGroupRightsHelpers.ts` are
  NOT touched by this plan (per the phase's directive (b) and `145-UI-SPEC.md`'s Scope Note) — the
  grant/revoke Switch flow for the pseudo-role's 3 actions reuses that existing, unmodified modal
  exactly like every other role.
