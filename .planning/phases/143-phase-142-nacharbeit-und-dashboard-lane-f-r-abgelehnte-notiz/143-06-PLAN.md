Original (pre-84a38c0f) `roleColorCode` implementation, restore verbatim from
`git show 6c35c59d:frontend/src/lib/roleColors.ts`:
```
import { FANSUB_GROUP_ROLE_OPTIONS } from '@/types/fansub'
const ROLE_CODE_BY_LABEL = new Map(FANSUB_GROUP_ROLE_OPTIONS.map((option) => [option.label, option.code]))
export function roleColorCode(roleLabel: string): string {
  const code = ROLE_CODE_BY_LABEL.get(roleLabel)
  if (code === 'techadmin') return 'admin'
  if (code === 'gfxler') return 'designer'
  return code ?? 'other'
}
```
