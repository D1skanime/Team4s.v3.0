<objective>
Build the idempotent, one-off backfill CLI command `backend/cmd/migrate-display-backfill` that
(a) adds a `display` media_files row for every existing media_asset that already has an
`original` but no `display`, reusing the exported static/animated display generators from
173-01/173-02, (b) special-cases pre-existing story images (no `media_files` rows at all — reuses
their existing file as the display source, per D-15's documented exception, no new file written),
and (c) migrates pre-existing flat-stored fansub logo/banner/group-media files into the
`/media/fansub/<group_id>/...` namespace introduced by 173-04, updating their stored paths.

Purpose: this is the SOLE new file/package the entire phase is permitted to add (per D-14's hard
constraint) — every other plan only modifies existing write paths. `173-RESEARCH.md`/
`173-PATTERNS.md` explicitly name `backend/cmd/migrate-preview-backfill/` (Phase 172) as the exact
structural template to copy, not `cmd/migrate-covers` (which lacks the testable `runBackfill`
split).

Output: a self-contained CLI binary runnable inside the existing backend container
(`DATABASE_URL`/`MEDIA_STORAGE_DIR`/`FFMPEG_PATH` env vars, same names the server already uses),
with a `DRY_RUN` flag and a structured stats summary, proven against a real Postgres fixture in
`backfill_test.go`.
</objective>

```go
func runBackfill(ctx context.Context, db *pgxpool.Pool, cfg Config) (*BackfillStats, error) {
	stats := &BackfillStats{}
	candidates, err := fetchBackfillCandidates(ctx, db)
	stats.TotalCandidates = len(candidates)
	for _, candidate := range candidates {
		if err := processBackfillCandidate(ctx, db, ..., candidate); err != nil {
			stats.Failed++; continue
		}
		stats.ProcessedOK++
	}
	return stats, nil
}

func fetchBackfillCandidates(ctx context.Context, db *pgxpool.Pool) ([]backfillCandidate, error) {
	rows, err := db.Query(ctx, `
		SELECT DISTINCT ON (ts.id) ts.id, t.anime_id, src.output_path, src.duration_seconds
		FROM theme_segments ts
		JOIN themes t ON t.id = ts.theme_id
		JOIN theme_segment_render_cache src ON src.theme_segment_id = ts.id
		WHERE ts.auto_preview_media_asset_id IS NULL
		  AND src.status = 'ready' AND src.invalidated_at IS NULL AND src.duration_seconds IS NOT NULL
		ORDER BY ts.id, src.completed_at DESC NULLS LAST, src.id DESC
	`)
	// ...
}

// Race-safe idempotent write guard pattern used downstream:
tag, err := db.Exec(ctx, `
	UPDATE theme_segments SET auto_preview_media_asset_id = $1
	WHERE id = $2 AND auto_preview_media_asset_id IS NULL
`, assetID, candidate.SegmentID)
if tag.RowsAffected() == 0 { /* someone else already processed it concurrently, clean up and skip */ }
```
This phase's equivalent "primary idempotency guard" query (verified against the live
`media_files` schema, migration 0026, `variant` is a free `VARCHAR(50)`):
```sql
SELECT DISTINCT ON (ma.id) ma.id, mf_orig.path, mf_orig.width, mf_orig.height, ma.mime_type
FROM media_assets ma
JOIN media_files mf_orig ON mf_orig.media_id = ma.id AND mf_orig.variant = 'original' AND mf_orig.status = 'ready'
LEFT JOIN media_files mf_display ON mf_display.media_id = ma.id AND mf_display.variant = 'display'
WHERE mf_display.id IS NULL
ORDER BY ma.id
```
and the race-safe write:
```sql
INSERT INTO media_files (media_id, variant, path, width, height, size, status)
SELECT $1, 'display', $2, $3, $4, $5, 'ready'
WHERE NOT EXISTS (SELECT 1 FROM media_files WHERE media_id = $1 AND variant = 'display')
```
