func animeCoverImageSelectSQL(animeAlias string) string {
	return fmt.Sprintf(`
		COALESCE(
			NULLIF(BTRIM(%s.cover_resolved_url), ''),
			poster.file_path,
			CASE
				WHEN %s.source LIKE 'jellyfin:%%' AND BTRIM(REPLACE(%s.source, 'jellyfin:', '')) <> ''

			END

}
