Weg-1 adapter sketch:
  uploadFiles: (files, onProgress) => uploadReleaseVersionMedia({ versionId, category, files,
    onProgress: (_i, pct) => onProgress(pct) }).then(r => r.results)
  readyResultId: (result) => result.release_version_media_id ?? null
  isReady: (result) => result.status === 'ready' && typeof result.release_version_media_id === 'number'
  readErrorFromResult: (result) => result.error_code || 'Upload fehlgeschlagen.'
  patchMetadata: (resultId, patch) => patchReleaseVersionMediaItem(versionId, resultId, {
    ...(patch.title ? { title: patch.title } : {}),
    ...(patch.caption ? { caption: patch.caption } : {}),
    // source_revision threading: the adapter must keep its OWN closure-local map of

    // updated from each patch response's returned source_revision) since useUploadQueue's own
    // UploadQueueItem shape does not carry this Weg-1-specific field — see Task 1's action for the
    // exact mechanism.
  }).then(() => ({ ok: true }), (e) => ({ ok: false, message: e instanceof Error ? e.message : 'Fehler' }))
