<!-- Weg-2 adapter to build in this plan, using the D-14 field shipped in plan 180-03: -->
uploadFansubGroupMedia({ fansubID, files, category, visibilityCode: 'private',
  reviewStatusCode: 'in_review', onProgress }) -> { results: [{ client_file_name, status,
  media_asset_id?, fansub_group_media_id?, preview_url?, thumbnail_url?, original_url?,
  error_code?, message? }] }
patchFansubMediaReview(fansubId, mediaId, patch: { title?, description?, alt_text?, visibility?,
  review_status?, category?, sort_order? }) -> { message: string }
  NOTE: the patch field is `description`, NOT `caption` — FansubMediaReviewPatch has no `caption`
  field (frontend/src/lib/api/admin/fansub-media.ts:220-228). The Weg-2 adapter's patchMetadata
  must map the hook's generic `patch.caption` onto `description` when calling
  patchFansubMediaReview — this field-name translation is the adapter's job, never the shared
  hook's.
  NOTE: unlike Weg 1, this endpoint carries no `source_revision`/optimistic-concurrency concept —
  do not invent one; the adapter's patchMetadata can be a direct, unconditional call.

Weg-2 adapter sketch:
  uploadFiles: (files, onProgress) => uploadFansubGroupMedia({ fansubID: fansubId, files, category:
    uploadCategory, visibilityCode: 'private', reviewStatusCode: 'in_review', onProgress }).then(r
    => r.results)
  readyResultId: (result) => result.fansub_group_media_id ?? null
  isReady: (result) => result.status === 'ready' && typeof result.fansub_group_media_id === 'number'
  readErrorFromResult: (result) => result.message || result.error_code || 'Upload fehlgeschlagen.'
  patchMetadata: (resultId, patch) => patchFansubMediaReview(fansubId, resultId, {
    ...(patch.title ? { title: patch.title } : {}),
    ...(patch.caption ? { description: patch.caption } : {}),
  }).then(() => ({ ok: true }), (e) => ({ ok: false, message: e instanceof Error ? e.message : 'Fehler' }))
