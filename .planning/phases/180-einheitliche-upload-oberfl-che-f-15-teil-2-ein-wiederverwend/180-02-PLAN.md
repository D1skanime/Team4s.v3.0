<!-- The adapter contract this hook is parametrized over. Weg-specific adapters are built in
     plans 180-05 (Weg 1) and 180-07 (Weg 2); this plan only defines + consumes the shape. -->

export interface UploadAdapter<TResult> {
  // Called once per startUpload/retryUpload batch. onProgress reports ONE overall percent for
  // the whole multipart request (authorizedUploadXhr's real behavior — frontend/src/lib/api/client.ts:963-1037
  // — there is no true per-file progress from the backend today; do not invent one).
  uploadFiles: (files: File[], onProgress: (percent: number) => void) => Promise<TResult[]>
  // Extracts the ready-state relation ID from one raw per-file result, or null if the file failed.
  // Weg 1: result => result.release_version_media_id ?? null
  // Weg 2 (after 180-03/D-14 ships): result => result.fansub_group_media_id ?? null
  readyResultId: (result: TResult) => number | null
  // Reads result.status === 'ready' (string literal differs slightly between the two backends'
  // response shapes — the adapter normalizes this, the hook never inspects raw status strings).
  isReady: (result: TResult) => boolean
  // Reads a free-text error/message off a failed result for display.
  readErrorFromResult: (result: TResult) => string | null
  // Optional metadata patch-after-upload step (title/caption). Weg 1 maps caption->caption via
  // patchReleaseVersionMediaItem; Weg 2 maps caption->description via patchFansubMediaReview
  // (FansubMediaReviewPatch has no `caption` field, only `description` — confirmed in
  // frontend/src/lib/api/admin/fansub-media.ts:220-228). The adapter owns this field-name mapping,
  // not the hook.
  patchMetadata?: (resultId: number, patch: { title?: string; caption?: string }) => Promise<{ ok: true } | { ok: false; message: string }>
}

export interface UploadQueueItem {
  file: File
  status: 'idle' | 'uploading' | 'processing' | 'ready' | 'failed'
  progress: number
  errorMessage: string | null
  resultId: number | null
  title?: string
  caption?: string
}

export interface UploadRunResult {
  items: UploadQueueItem[]
  allSucceeded: boolean
}
