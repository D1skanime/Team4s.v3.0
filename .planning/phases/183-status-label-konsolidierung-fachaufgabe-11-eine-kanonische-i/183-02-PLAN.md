```typescript
function claimStatusLabel(status: string): string {
  switch (status) {
    case 'active':
    case 'verified':
      return 'Verifiziert'   // D-03: Wort wird zu 'Genehmigt' geändert
    case 'open':
    case 'pending':
      return 'Offen'
    case 'rejected':
    case 'denied':
      return 'Abgelehnt'
    default:
      return status
  }
}
function claimStatusVariant(status: string): 'success' | 'warning' | 'danger' | 'neutral' {
  switch (status) {
    case 'active':
    case 'verified':
      return 'success'
    case 'open':
    case 'pending':
      return 'warning'
    case 'rejected':
    case 'denied':
      return 'danger'
    default:
      return 'neutral'
  }
}
```
Required exported names (per D-03, verbatim): `claimStatusVariant`, `claimStatusLabel`. Final
locked mapping: `verified`/`active` -> `'Genehmigt'`/`'success'`; `pending`/`open` ->
`'Offen'`/`'warning'`; `rejected`/`denied` -> `'Abgelehnt'`/`'danger'`; unbekannt ->
Rohwert/`'neutral'`.

`Badge` from `@/components/ui` accepts `variant?: 'neutral' | 'success' | 'warning' | 'danger' |
'info' | 'muted'` — the helper's 4-value return type is a subset, no type mismatch.
