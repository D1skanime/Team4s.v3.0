<objective>
Reduce the public group-profile load path from 12 static queries to 7 by hydrating only the fields the
public page actually reads (B1), eliminating the duplicate `fansub_group_links` query (B2), and
dropping the domain-projection's never-rendered contributors query (B3). This is backend/query work,
kept in its own plan/wave separate from all UI-touching plans (D09/critical_sequencing_constraint #2)
and touching zero files any UI plan touches.

Purpose: `GetGroupBySlug`/`hydrateFansubGroup` currently attach 5 count queries (4 of which the public
page never reads) and 1 link query that is then queried a SECOND time via `ListGroupLinks` — the exact
duplication named in the audit. `DomainProjectionRepository.GetFansubGroupDomainProjection` loads
contributors that `FansubContributorsSection` never renders anywhere in the component tree. Per D07,
`public-profile` and `domain-projection` stay technically separate endpoints — this plan trims each
independently without merging them.
Output: Additive, non-destructive public-specific load-path helpers in `fansub_repository.go` (the
existing `GetGroupBySlug`/`GetGroupByID`/`hydrateFansubGroup`/`attachGroupCounts`/`attachGroupLinks`
stay byte-for-byte unchanged for their other 4+ callers); a trimmed `GetFansubGroupDomainProjection`;
a new guarded-Postgres test file proving both changes behaviorally.
</objective>

```go
func (r *FansubRepository) GetGroupBySlug(ctx context.Context, slug string) (*models.FansubGroup, error) {
	query := `
		SELECT
			id, slug, name, logo_id, banner_id, logo_url, banner_url,
			founded_year, dissolved_year, closed_year, status, 'group' AS group_type, website_url, discord_url, irc_url, country,
			created_at, updated_at
		FROM fansub_groups
		WHERE slug = $1
	`
	var item models.FansubGroup
	if err := r.db.QueryRow(ctx, query, slug).Scan(
		&item.ID, &item.Slug, &item.Name, &item.LogoID, &item.BannerID, &item.LogoURL, &item.BannerURL,
		&item.FoundedYear, &item.DissolvedYear, &item.ClosedYear, &item.Status, &item.GroupType,
		&item.WebsiteURL, &item.DiscordURL, &item.IrcURL, &item.Country, &item.CreatedAt, &item.UpdatedAt,
	); errors.Is(err, pgx.ErrNoRows) {
		return nil, ErrNotFound
	} else if err != nil {
		return nil, fmt.Errorf("get fansub group %q: %w", slug, err)
	}
	if err := r.hydrateFansubGroup(ctx, &item); err != nil {   // <- getPublicGroupBase omits this call
		return nil, err
	}
	return &item, nil
}
```
