    const STORY_IDS = [
      'zusammengesetzte-bausteine-release-versionszeile--mit-gruppenlogo',
      'zusammengesetzte-bausteine-release-versionszeile--ohne-logo',
      'zusammengesetzte-bausteine-release-versionszeile--koop',
      'zusammengesetzte-bausteine-release-versionszeile--mit-allen-extras',
    ]
    const VIEWPORTS = [
      { label: 'desktop-1440', width: 1440, height: 900 },
      { label: 'mobile-390', width: 390, height: 844 },
    ]

    async function screenshot(browser, baseUrl, storyId, viewport) {
      const page = await browser.newPage({ viewport: { width: viewport.width, height: viewport.height } })
      await page.goto(`${baseUrl}/iframe.html?id=${storyId}&viewMode=story`, { waitUntil: 'networkidle' })
      const buffer = await page.screenshot({ fullPage: true })
      await page.close()
      return buffer
    }

    async function main() {
      const [baselineUrl, afterUrl] = process.argv.slice(2)
      const browser = await chromium.launch({ headless: true })
      let anyDiff = false

      for (const storyId of STORY_IDS) {
        for (const viewport of VIEWPORTS) {
          const [beforeBuf, afterBuf] = await Promise.all([
            screenshot(browser, baselineUrl, storyId, viewport),
            screenshot(browser, afterUrl, storyId, viewport),
          ])
          const [beforeRaw, afterRaw] = await Promise.all([
            sharp(beforeBuf).raw().toBuffer({ resolveWithObject: true }),
            sharp(afterBuf).raw().toBuffer({ resolveWithObject: true }),
          ])
          if (beforeRaw.info.width !== afterRaw.info.width || beforeRaw.info.height !== afterRaw.info.height) {
            anyDiff = true
            console.error(`DIMENSION MISMATCH ${storyId} ${viewport.label}: before=${beforeRaw.info.width}x${beforeRaw.info.height} after=${afterRaw.info.width}x${afterRaw.info.height}`)
            continue
          }
          const { width, height } = beforeRaw.info
          const diffPixels = pixelmatch(beforeRaw.data, afterRaw.data, null, width, height, { threshold: 0 })
          console.log(`${storyId} ${viewport.label}: diffPixels=${diffPixels}/${width * height}`)
          if (diffPixels !== 0) anyDiff = true
        }
      }

      await browser.close()
      if (anyDiff) { console.error('ZERO_PIXEL_DIFF_FAILED'); process.exit(1) }
      console.log('ZERO_PIXEL_DIFF_OK')
    }

    main().catch((error) => { console.error(error); process.exit(1) })
