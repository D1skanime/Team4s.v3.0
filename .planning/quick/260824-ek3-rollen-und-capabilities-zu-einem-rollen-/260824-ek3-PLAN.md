Existing `CATEGORY_ORDER` + sort logic currently duplicated only inside `RoleCapabilityDetail.tsx`
(lines 11-19, 68-75) -- Task 1 extracts it into `capabilityCategories.ts` as an exported
`sortCategories(categories: string[]): string[]` so `RolesClient`'s "open the first category on role
change" logic (Task 3, GAP-05/D-08 "erste Kategorie offen") uses the identical ordering, not a second
divergent copy:

```typescript
const CATEGORY_ORDER = [
  'gruppe', 'gruppenmedien', 'gruppenseite', 'projekt', 'rechteverwaltung', 'release', 'review',
]

export function sortCategories(categories: string[]): string[] {
  return [...categories].sort((a, b) => {
    const ai = CATEGORY_ORDER.indexOf(a)
    const bi = CATEGORY_ORDER.indexOf(b)
    if (ai === -1 && bi === -1) return a.localeCompare(b)
    if (ai === -1) return 1
    if (bi === -1) return -1
    return ai - bi
  })
}
```
